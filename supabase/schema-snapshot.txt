CREATE OR REPLACE FUNCTION public.anonymize_expired_profiles()
 RETURNS TABLE(profile_id uuid, old_avatar_url text)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
begin
  return query
  with expired as (
    select id, avatar_url
    from public.profiles
    where is_active = false
      and deactivated_at is not null
      and deactivated_at <= now() - interval '90 days'
      and anonymized_at is null
    for update
  ),
  updated as (
    update public.profiles p
    set
      first_name = 'Verwijderd',
      last_name = 'lid',
      email = 'verwijderd-' || p.id || '@voc-ledenportaal.invalid',
      phone = null,
      job_title = null,
      bio = null,
      avatar_url = null,
      linkedin_url = null,
      anonymized_at = now(),
      pending_avatar_cleanup = expired.avatar_url
    from expired
    where p.id = expired.id
    returning p.id
  )
  select expired.id, expired.avatar_url from expired;
end;
$function$

CREATE OR REPLACE FUNCTION public.anonymize_profile_now(p_id uuid)
 RETURNS TABLE(profile_id uuid, old_avatar_url text)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
begin
  return query
  with target as (
    select id, avatar_url
    from public.profiles
    where id = p_id and anonymized_at is null
    for update
  ),
  updated as (
    update public.profiles p
    set
      first_name = 'Verwijderd',
      last_name = 'lid',
      email = 'verwijderd-' || p.id || '@voc-ledenportaal.invalid',
      phone = null,
      job_title = null,
      bio = null,
      avatar_url = null,
      linkedin_url = null,
      is_active = false,
      deactivated_at = coalesce(p.deactivated_at, now()),
      anonymized_at = now(),
      pending_avatar_cleanup = target.avatar_url
    from target
    where p.id = target.id
    returning p.id
  )
  select target.id, target.avatar_url from target;
end;
$function$

CREATE OR REPLACE FUNCTION public.claim_newsletter_contact_recipients(p_communication_id uuid)
 RETURNS TABLE(send_id uuid, contact_id uuid, email text)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_filter text;
begin
  if not (public.is_board() or auth.role() = 'service_role') then
    raise exception 'not authorized';
  end if;

  select recipient_filter into v_filter from public.communications where id = p_communication_id;

  insert into public.communication_contact_sends (communication_id, contact_id)
  select p_communication_id, ct.id
  from public.contacts ct
  where not ct.opted_out
    and (
      (v_filter = 'lijsten' and exists (
        select 1
        from public.communication_recipient_lists crl
        join public.contact_list_contacts clc on clc.list_id = crl.list_id
        where crl.communication_id = p_communication_id and clc.contact_id = ct.id
      ))
      or exists (
        select 1 from public.communication_recipient_contacts crc
        where crc.communication_id = p_communication_id and crc.contact_id = ct.id
      )
    )
  on conflict (communication_id, contact_id) do nothing;

  update public.communications
  set total_recipients = (
    (select count(*) from public.notifications where communication_id = p_communication_id)
    + (select count(*) from public.communication_contact_sends where communication_id = p_communication_id)
  )
  where id = p_communication_id;

  return query
  select s.id, s.contact_id, ct.email
  from public.communication_contact_sends s
  join public.contacts ct on ct.id = s.contact_id
  where s.communication_id = p_communication_id
    and s.emailed_at is null;
end;
$function$

CREATE OR REPLACE FUNCTION public.claim_newsletter_deliveries(p_communication_id uuid, p_kind text)
 RETURNS TABLE(delivery_id uuid, profile_id uuid, contact_id uuid, email text, attempt_id uuid)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_filter text;
  v_attempt_id uuid;
begin
  if not (public.is_board() or auth.role() = 'service_role') then
    raise exception 'not authorized';
  end if;
  if p_kind not in ('eerste_verzending', 'nasturen', 'opnieuw_proberen') then
    raise exception 'ongeldig soort verzendpoging: %', p_kind;
  end if;

  select recipient_filter into v_filter from public.communications where id = p_communication_id;
  if v_filter is null then
    raise exception 'nieuwsbrief niet gevonden';
  end if;

  insert into public.communication_send_attempts (communication_id, kind, started_by)
  values (p_communication_id, p_kind, auth.uid())
  returning id into v_attempt_id;

  insert into public.communication_deliveries (communication_id, profile_id, email, status, failure_reason)
  select
    p_communication_id,
    p.id,
    lower(trim(p.email)),
    case when p.email_campaigns then 'wachten' else 'overgeslagen' end,
    case when p.email_campaigns then null else 'afgemeld' end
  from public.profiles p
  where p.is_active
    and (
      v_filter = 'alle_actieve_leden'
      or exists (
        select 1 from public.communication_recipient_lists crl
        join public.contact_list_profiles clp on clp.list_id = crl.list_id
        where crl.communication_id = p_communication_id and clp.profile_id = p.id
      )
      or exists (
        select 1 from public.communication_recipient_profiles crp
        where crp.communication_id = p_communication_id and crp.profile_id = p.id
      )
    )
  on conflict (communication_id, email) do nothing;

  insert into public.communication_deliveries (communication_id, contact_id, email, status, failure_reason)
  select
    p_communication_id,
    ct.id,
    lower(trim(ct.email)),
    case when ct.opted_out then 'overgeslagen' else 'wachten' end,
    case when ct.opted_out then 'afgemeld' else null end
  from public.contacts ct
  where (
      v_filter = 'lijsten' and exists (
        select 1 from public.communication_recipient_lists crl
        join public.contact_list_contacts clc on clc.list_id = crl.list_id
        where crl.communication_id = p_communication_id and clc.contact_id = ct.id
      )
    )
    or exists (
      select 1 from public.communication_recipient_contacts crc
      where crc.communication_id = p_communication_id and crc.contact_id = ct.id
    )
  on conflict (communication_id, email) do nothing;

  update public.communications
  set total_recipients = (select count(*) from public.communication_deliveries where communication_id = p_communication_id)
  where id = p_communication_id;

  update public.communication_deliveries
  set attempt_id = v_attempt_id, status = 'wachten', failure_reason = null
  where communication_id = p_communication_id
    and status in ('wachten', 'mislukt');

  return query
  select d.id, d.profile_id, d.contact_id, d.email, d.attempt_id

CREATE OR REPLACE FUNCTION public.claim_newsletter_recipients(p_communication_id uuid)
 RETURNS TABLE(notification_id uuid, profile_id uuid, email text)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_filter text;
begin
  if not (public.is_board() or auth.role() = 'service_role') then
    raise exception 'not authorized';
  end if;

  select recipient_filter into v_filter from public.communications where id = p_communication_id;

  insert into public.notifications (profile_id, type, title, communication_id, channel_push_allowed, channel_email_allowed)
  select p.id, 'newsletter', c.subject, c.id, false, false
  from public.profiles p
  cross join public.communications c
  where c.id = p_communication_id
    and p.is_active
    and p.email_campaigns
    and (
      v_filter = 'alle_actieve_leden'
      or (v_filter = 'lijsten' and exists (
        select 1
        from public.communication_recipient_lists crl
        join public.contact_list_profiles clp on clp.list_id = crl.list_id
        where crl.communication_id = c.id and clp.profile_id = p.id
      ))
      or exists (
        select 1 from public.communication_recipient_profiles crp
        where crp.communication_id = c.id and crp.profile_id = p.id
      )
    )
  on conflict (communication_id, profile_id) where communication_id is not null do nothing;

  -- Niet langer "only if null" (0067): de doelgroep kan na een eerste
  -- verzending nog groeien (nasturen), dus total_recipients telt bij elke
  -- claim-aanroep opnieuw de volledige, cumulatieve bedoelde ontvangerslijst
  -- (leden + contacten) na.
  update public.communications
  set total_recipients = (
    (select count(*) from public.notifications where communication_id = p_communication_id)
    + (select count(*) from public.communication_contact_sends where communication_id = p_communication_id)
  )
  where id = p_communication_id;

  return query
  select n.id, n.profile_id, p.email
  from public.notifications n
  join public.profiles p on p.id = n.profile_id
  where n.communication_id = p_communication_id
    and n.emailed_at is null;
end;
$function$

CREATE OR REPLACE FUNCTION public.dispatch_mention_notifications(p_content text, p_author_id uuid, p_context text)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_author_name text;
  v_match text[];
  v_profile_ids uuid[] := '{}';
  v_company_ids uuid[] := '{}';
  v_target_ids uuid[];
begin
  if p_content is null then
    return;
  end if;

  for v_match in
    select regexp_matches(p_content, '@\[[^\]]+\]\((profiel|bedrijf):([0-9a-fA-F-]{36})\)', 'g')
  loop
    if v_match[1] = 'profiel' then
      v_profile_ids := array_append(v_profile_ids, v_match[2]::uuid);
    else
      v_company_ids := array_append(v_company_ids, v_match[2]::uuid);
    end if;
  end loop;

  if array_length(v_profile_ids, 1) is null and array_length(v_company_ids, 1) is null then
    return;
  end if;

  select first_name || ' ' || last_name into v_author_name
    from public.profiles where id = p_author_id;

  select array_agg(distinct target) into v_target_ids
  from (
    select p.id as target
    from public.profiles p
    where p.id = any(v_profile_ids) and p.is_active and p.id <> p_author_id

    union

    select cm.profile_id as target
    from public.company_members cm
    join public.profiles p on p.id = cm.profile_id
    where cm.company_id = any(v_company_ids) and p.is_active and cm.profile_id <> p_author_id
  ) t;

  if v_target_ids is null then
    return;
  end if;

  insert into public.notifications (profile_id, type, title, body, link)
  select target, 'feed_mention', 'Je bent genoemd in de feed',
    coalesce(v_author_name, 'Iemand') || ' heeft je genoemd in ' || p_context || '.',
    '/'
  from unnest(v_target_ids) as target;
end;
$function$

CREATE OR REPLACE FUNCTION public.enforce_access_request_rate_limit()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_normalized_email text := lower(trim(new.email));
  v_recent_attempts integer;
  v_recent_total integer;
begin
  new.created_at := now();
  new.status := 'pending';

  select count(*) into v_recent_attempts
    from public.access_requests
    where lower(email) = v_normalized_email
      and created_at > now() - interval '15 minutes';

  if v_recent_attempts >= 5 then
    raise exception 'Te veel aanvragen met dit e-mailadres. Probeer het over een kwartier opnieuw.';
  end if;

  -- Los van het per-e-mailadres-limiet hierboven: dit formulier is volledig
  -- anoniem bereikbaar, dus een script dat telkens een ander e-mailadres
  -- gebruikt, omzeilt dat limiet moeiteloos. Een algemene bovengrens per
  -- tijdvak (onafhankelijk van welk e-mailadres) maakt dat soort
  -- grootschalig misbruik ook onmogelijk, zonder het legitieme, lage
  -- volume van dit formulier te raken.
  select count(*) into v_recent_total
    from public.access_requests
    where created_at > now() - interval '1 hour';

  if v_recent_total >= 30 then
    raise exception 'Er zijn momenteel te veel aanvragen binnengekomen. Probeer het later opnieuw.';
  end if;

  return new;
end;
$function$

CREATE OR REPLACE FUNCTION public.enforce_activity_registration_rules()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_deadline timestamptz;
  v_max integer;
  v_status text;
  v_confirmed_count integer;
begin
  new.attended := false;

  select registration_deadline, max_participants, status
    into v_deadline, v_max, v_status
    from public.activities
    where id = new.activity_id
    for update;

  if v_status = 'cancelled' then
    raise exception 'Deze activiteit is afgelast.';
  end if;

  if v_deadline is not null and now() > v_deadline then
    raise exception 'De aanmelddeadline voor deze activiteit is verstreken.';
  end if;

  if v_max is not null then
    select
      (select count(*) from public.activity_registrations where activity_id = new.activity_id and not is_waitlisted)
      + (select count(*) from public.public_activity_registrations where activity_id = new.activity_id)
      into v_confirmed_count;

    new.is_waitlisted := v_confirmed_count >= v_max;
  else
    new.is_waitlisted := false;
  end if;

  return new;
end;
$function$

CREATE OR REPLACE FUNCTION public.finalize_newsletter_send(p_communication_id uuid)
 RETURNS TABLE(total integer, sent integer)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_total integer;
  v_sent integer;
begin
  if not (public.is_board() or auth.role() = 'service_role') then
    raise exception 'not authorized';
  end if;

  select
    (select count(*) from public.notifications where communication_id = p_communication_id)
    + (select count(*) from public.communication_contact_sends where communication_id = p_communication_id),
    (select count(*) from public.notifications where communication_id = p_communication_id and emailed_at is not null)
    + (select count(*) from public.communication_contact_sends where communication_id = p_communication_id and emailed_at is not null)
  into v_total, v_sent;

  update public.communications
  set status = case when v_total > 0 and v_sent >= v_total then 'verzonden' else 'verzenden_mislukt' end,
      sent_at = case when v_total > 0 and v_sent >= v_total and sent_at is null then now() else sent_at end
  where id = p_communication_id;

  return query select v_total, v_sent;
end;
$function$

CREATE OR REPLACE FUNCTION public.finalize_newsletter_send_v2(p_communication_id uuid)
 RETURNS TABLE(total integer, sent integer, failed integer, skipped integer)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_total integer;
  v_sent integer;
  v_failed integer;
  v_skipped integer;
begin
  if not (public.is_board() or auth.role() = 'service_role') then
    raise exception 'not authorized';
  end if;

  select
    count(*),
    count(*) filter (where status = 'verstuurd'),
    count(*) filter (where status = 'mislukt'),
    count(*) filter (where status = 'overgeslagen')
  into v_total, v_sent, v_failed, v_skipped
  from public.communication_deliveries
  where communication_id = p_communication_id;

  update public.communications
  set status = case when v_total > 0 and (v_sent + v_skipped) >= v_total then 'verzonden' else 'verzenden_mislukt' end,
      sent_at = case
        when v_total > 0 and (v_sent + v_skipped) >= v_total and sent_at is null then now()
        else sent_at
      end
  where id = p_communication_id;

  return query select v_total, v_sent, v_failed, v_skipped;
end;
$function$

CREATE OR REPLACE FUNCTION public.generate_unique_company_slug(p_name text)
 RETURNS text
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_base text;
  v_candidate text;
  v_suffix int := 1;
begin
  v_base := nullif(trim(both '-' from regexp_replace(lower(p_name), '[^a-z0-9]+', '-', 'g')), '');
  v_base := coalesce(v_base, 'bedrijf');
  v_candidate := v_base;
  while exists (select 1 from public.companies where slug = v_candidate) loop
    v_suffix := v_suffix + 1;
    v_candidate := v_base || '-' || v_suffix;
  end loop;
  return v_candidate;
end;
$function$

CREATE OR REPLACE FUNCTION public.get_pending_email_notifications(p_limit integer DEFAULT 50)
 RETURNS TABLE(notification_id uuid, type text, title text, body text, link text, email text)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
begin
  update public.notifications n
  set emailed_at = now()
  from public.profiles p
  where n.profile_id = p.id
    and n.emailed_at is null
    and n.communication_id is null
    and (
      not n.channel_email_allowed
      or n.type = 'new_member'
      or (n.type in ('new_activity', 'activity_reminder') and not p.email_activities)
      or (n.type in ('feed_comment', 'feed_mention') and not p.email_feed)
    );

  return query
  with claimed as (
    select n.id
    from public.notifications n
    where n.emailed_at is null
      and n.communication_id is null
      and (n.email_claimed_at is null or n.email_claimed_at < now() - interval '10 minutes')
    order by n.created_at
    limit p_limit
    for update of n skip locked
  ),
  leased as (
    update public.notifications n
    set email_claimed_at = now()
    from claimed c
    where n.id = c.id
    returning n.id, n.type, n.title, n.body, n.link, n.profile_id
  )
  select l.id, l.type, l.title, l.body, l.link, p.email
  from leased l
  join public.profiles p on p.id = l.profile_id;
end;
$function$

CREATE OR REPLACE FUNCTION public.get_pending_push_notifications(p_limit integer DEFAULT 50)
 RETURNS TABLE(notification_id uuid, type text, title text, body text, link text, endpoint text, p256dh text, auth text)
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
begin
  update public.notifications n
  set pushed_at = now()
  from public.profiles p
  where n.profile_id = p.id
    and n.pushed_at is null
    and n.communication_id is null
    and (
      not n.channel_push_allowed
      or n.type = 'new_member'
      or (n.type in ('new_activity', 'activity_reminder') and not p.push_activities)
      or (n.type in ('feed_comment', 'feed_mention') and not p.push_feed)
    );

  return query
  with claimed as (
    select n.id, n.type, n.title, n.body, n.link, n.profile_id
    from public.notifications n
    where n.pushed_at is null
      and n.communication_id is null
      and exists (select 1 from public.push_subscriptions s where s.profile_id = n.profile_id)
    order by n.created_at
    limit p_limit
    for update of n skip locked
  ),
  marked as (
    update public.notifications n
    set pushed_at = now()
    from claimed c
    where n.id = c.id
    returning n.id, n.type, n.title, n.body, n.link, n.profile_id
  )
  select m.id, m.type, m.title, m.body, m.link, s.endpoint, s.p256dh, s.auth
  from marked m
  join public.push_subscriptions s on s.profile_id = m.profile_id;
end;
$function$

CREATE OR REPLACE FUNCTION public.handle_company_membership_decision()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_cfg record;
begin
  if new.status = 'approved' and old.status is distinct from 'approved' then
    delete from public.company_members
    where profile_id = new.profile_id and company_id <> new.company_id;

    insert into public.company_members (company_id, profile_id)
    values (new.company_id, new.profile_id)
    on conflict do nothing;
  end if;

  if new.status in ('approved', 'rejected') and old.status is distinct from new.status then
    new.decided_at = now();

    select * into v_cfg from public.notification_dispatch_config('company_membership_decision');
    if v_cfg.enabled then
      insert into public.notifications (profile_id, type, title, body, link, channel_push_allowed, channel_email_allowed, channel_inapp_allowed)
      values (
        new.profile_id,
        'company_membership_decision',
        case when new.status = 'approved' then 'Bedrijfskoppeling goedgekeurd' else 'Bedrijfskoppeling afgewezen' end,
        case
          when new.status = 'approved' then 'Je aanvraag om aan dit bedrijf gekoppeld te worden is goedgekeurd.'
          when new.decision_reason is not null and new.decision_reason <> ''
            then 'Je aanvraag om aan dit bedrijf gekoppeld te worden is afgewezen. Reden: ' || new.decision_reason
          else 'Je aanvraag om aan dit bedrijf gekoppeld te worden is afgewezen.'
        end,
        '/bedrijven/' || (select slug from public.companies where id = new.company_id),
        v_cfg.channel_push, v_cfg.channel_email, v_cfg.channel_inapp
      );
    end if;
  end if;

  return new;
end;
$function$

CREATE OR REPLACE FUNCTION public.handle_new_user()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_meta jsonb := new.raw_user_meta_data;
  v_token text := v_meta ->> 'invitation_token';
  v_invitation public.invitations;
  v_company_id uuid := nullif(v_meta ->> 'company_id', '')::uuid;
  v_new_company_name text := v_meta ->> 'new_company_name';
  v_created_company_id uuid;
  v_first_name text := coalesce(v_meta ->> 'first_name', '');
  v_last_name text := coalesce(v_meta ->> 'last_name', '');
begin
  if v_token is null then
    raise exception 'Registratie vereist een geldige uitnodiging.';
  end if;

  select * into v_invitation
    from public.invitations
    where token = v_token
    for update;

  if v_invitation is null
     or v_invitation.status <> 'pending'
     or v_invitation.expires_at is null
     or v_invitation.expires_at < now()
  then
    raise exception 'Deze uitnodiging is ongeldig of verlopen.';
  end if;

  if v_invitation.email is not null and lower(v_invitation.email) <> lower(new.email) then
    raise exception 'Dit e-mailadres komt niet overeen met de uitnodiging.';
  end if;

  insert into public.profiles (id, first_name, last_name, email, phone, job_title, role, slug)
  values (
    new.id,
    v_first_name,
    v_last_name,
    new.email,
    v_meta ->> 'phone',
    v_meta ->> 'job_title',
    v_invitation.role,
    public.generate_unique_profile_slug(v_first_name, v_last_name)
  );

  if v_new_company_name is not null and v_new_company_name <> '' then
    insert into public.companies (name, slug, industry, website, city)
    values (
      v_new_company_name,
      public.generate_unique_company_slug(v_new_company_name),
      v_meta ->> 'new_company_industry',
      v_meta ->> 'new_company_website',
      v_meta ->> 'new_company_city'
    )
    returning id into v_created_company_id;

    insert into public.company_members (company_id, profile_id) values (v_created_company_id, new.id);
  elsif v_company_id is not null then
    if v_invitation.company_id is not null and v_invitation.company_id = v_company_id then
      insert into public.company_members (company_id, profile_id) values (v_company_id, new.id);
    else
      insert into public.company_membership_requests (company_id, profile_id) values (v_company_id, new.id);
    end if;
  end if;

  update public.invitations
    set status = 'accepted', accepted_by = new.id, accepted_at = now()
    where id = v_invitation.id;

  return new;
end;
$function$

        when new.rejection_reason is not null and new.rejection_reason <> ''
          then 'Je activiteit "' || new.title || '" is afgewezen: ' || new.rejection_reason
        else 'Je activiteit "' || new.title || '" is afgewezen.'
      end,
      '/agenda/' || new.slug,

  select first_name || ' ' || last_name into v_applicant_name from public.profiles where id = new.profile_id;

    select profile_id as recipient_id from public.company_members where company_id = new.company_id

    end if;

    select first_name || ' ' || last_name into v_commenter_name
      from public.profiles where id = new.author_id;

    insert into public.notifications (profile_id, type, title, body, link, channel_push_allowed, channel_email_allowed, channel_inapp_allowed)
    values (
      v_post_author,
      'feed_comment',
      'Nieuwe reactie op je bericht',
      coalesce(v_commenter_name, 'Iemand') || ' heeft gereageerd op je bericht.',
      '/community?highlight=' || new.id,
      v_cfg.channel_push, v_cfg.channel_email, v_cfg.channel_inapp
    );
  end if;

  return new;
end;
$function$

CREATE OR REPLACE FUNCTION public.notify_report_resolved(p_report_id uuid, p_decision text)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_reporter_id uuid;
  v_cfg record;
begin
  if not public.is_board() then
    raise exception 'not authorized';
  end if;

  select reporter_id into v_reporter_id from public.feed_post_reports where id = p_report_id;
  if v_reporter_id is null then
    return;
  end if;

  select * into v_cfg from public.notification_dispatch_config('report_resolved');
  if not v_cfg.enabled then
    return;
  end if;

  insert into public.notifications (profile_id, type, title, body, channel_push_allowed, channel_email_allowed, channel_inapp_allowed)
  values (
    v_reporter_id,
    'report_resolved',
    'Je melding is behandeld',
    case
      when p_decision = 'deleted' then 'Het gemelde bericht is verwijderd.'
      else 'De melding is beoordeeld; het bericht blijft staan.'
    end,
    v_cfg.channel_push,
    v_cfg.channel_email,
    v_cfg.channel_inapp
  );
end;
$function$

CREATE OR REPLACE FUNCTION public.promote_next_waitlisted()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_next public.activity_registrations;
  v_title text;
  v_slug text;
  v_cfg record;
begin
  if old.is_waitlisted then
    return old;
  end if;

  select * into v_next
    from public.activity_registrations
    where activity_id = old.activity_id and is_waitlisted
    order by created_at
    limit 1
    for update;

  if v_next.id is null then
    return old;
  end if;

  update public.activity_registrations set is_waitlisted = false where id = v_next.id;

  select * into v_cfg from public.notification_dispatch_config('waitlist_promoted');
  if v_cfg.enabled then
    select title, slug into v_title, v_slug from public.activities where id = old.activity_id;

    insert into public.notifications (profile_id, type, title, body, link, channel_push_allowed, channel_email_allowed, channel_inapp_allowed)
    values (
      v_next.profile_id,
      'waitlist_promoted',
      'Je staat nu op de deelnemerslijst',
      'Er is een plek vrijgekomen voor "' || coalesce(v_title, 'de activiteit') || '" — je bent van de wachtlijst gehaald.',
      '/agenda/' || v_slug,
      v_cfg.channel_push, v_cfg.channel_email, v_cfg.channel_inapp
    );
  end if;

  return old;
end;
$function$

CREATE OR REPLACE FUNCTION public.record_manual_push_broadcast(p_reached_profile_ids uuid[], p_dead_endpoints text[], p_title text, p_body text)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
begin
  if not public.is_admin() then
    raise exception 'not authorized';
  end if;

  insert into public.notifications (profile_id, type, title, body, pushed_at)
  select unnest(p_reached_profile_ids), 'manual_broadcast', p_title, p_body, now();

  if array_length(p_dead_endpoints, 1) > 0 then
    insert into public.events (event_type, target_type, metadata)
    select 'push_unsubscribed', 'push_subscription', jsonb_build_object('endpoint', endpoint)
    from unnest(p_dead_endpoints) as endpoint;

    delete from public.push_subscriptions where endpoint = any(p_dead_endpoints);
  end if;
end;
$function$

CREATE OR REPLACE FUNCTION public.sync_prospect_from_public_registration()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
begin
  insert into public.prospects (email, name, company_name, last_seen_at)
  values (lower(new.email), new.name, new.company_name, now())
  on conflict (email) do update set
    name = excluded.name,
    company_name = coalesce(excluded.company_name, public.prospects.company_name),
    last_seen_at = now();
  return new;
end;

---
CREATE OR REPLACE FUNCTION public.upsert_public_activity_registration(p_activity_id uuid, p_name text, p_email text, p_company_name text)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_deadline timestamptz;
  v_max integer;
  v_status text;
  v_allow boolean;
  v_exists boolean;
  v_confirmed_count integer;
  v_normalized_email text := lower(trim(p_email));
  v_recent_attempts integer;
begin
  select count(*) into v_recent_attempts
    from public.events
    where event_type = 'public_activity_registration_attempted'
      and created_at > now() - interval '15 minutes'
      and metadata ->> 'email' = v_normalized_email;

  if v_recent_attempts >= 5 then
    raise exception 'Te veel pogingen met dit e-mailadres. Probeer het over een kwartier opnieuw.';
  end if;

  insert into public.events (event_type, metadata)
  values ('public_activity_registration_attempted', jsonb_build_object('email', v_normalized_email));

  select registration_deadline, max_participants, status, allow_public_registration
    into v_deadline, v_max, v_status, v_allow
    from public.activities
    where id = p_activity_id
    for update;

  if v_status is distinct from 'approved' or not coalesce(v_allow, false) then
    raise exception 'Deze activiteit staat niet open voor aanmelden.';
  end if;

  select exists (
    select 1 from public.public_activity_registrations
    where activity_id = p_activity_id and email = p_email
  ) into v_exists;

  if not v_exists then
    if v_deadline is not null and now() > v_deadline then
      raise exception 'De aanmelddeadline voor deze activiteit is verstreken.';
    end if;

    if v_max is not null then
      select
        (select count(*) from public.activity_registrations where activity_id = p_activity_id and not is_waitlisted)
        + (select count(*) from public.public_activity_registrations where activity_id = p_activity_id)
        into v_confirmed_count;

      if v_confirmed_count >= v_max then
        raise exception 'Deze activiteit heeft het maximum aantal deelnemers bereikt.';
      end if;
    end if;
  end if;

  insert into public.public_activity_registrations (activity_id, name, email, company_name)
  values (p_activity_id, p_name, p_email, p_company_name)
  on conflict (activity_id, email)
  do update set name = excluded.name, company_name = excluded.company_name;
end;
$function$
